-- ========================================
-- QR 스탬프 투어 시스템 - 완전판 (오타 수정)
-- ========================================

-- ========================================
-- 1. events 테이블
-- ========================================
CREATE TABLE IF NOT EXISTS events (
    event_id SERIAL PRIMARY KEY,
    event_name VARCHAR(100) NOT NULL,
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    required_stamps INTEGER DEFAULT 14
);

-- ========================================
-- 2. users 테이블 (role 포함)
-- ========================================
CREATE TABLE IF NOT EXISTS users (
    student_id VARCHAR(20) PRIMARY KEY,
    password_hash VARCHAR(255) NOT NULL,
    name VARCHAR(50) NOT NULL,
    phone_number VARCHAR(20) UNIQUE NOT NULL,
    dept_name VARCHAR(50) NOT NULL,
    grade INTEGER NOT NULL,
    gender VARCHAR(10) NOT NULL,
    role VARCHAR(20) NOT NULL DEFAULT 'user',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT check_role CHECK (role IN ('user', 'admin'))
);

-- ========================================
-- 3. booths 테이블
-- ========================================
CREATE TABLE IF NOT EXISTS booths (
    booth_id SERIAL PRIMARY KEY,
    event_id INTEGER REFERENCES events(event_id),
    booth_name VARCHAR(100) NOT NULL,
    location_name VARCHAR(100),
    latitude NUMERIC(10, 8) NOT NULL,
    longitude NUMERIC(11, 8) NOT NULL,
    qr_code VARCHAR(50) UNIQUE,
    qr_pin VARCHAR(10),
    icon VARCHAR(10)
);

-- ========================================
-- 4. stamps 테이블
-- ========================================
CREATE TABLE IF NOT EXISTS stamps (
    stamp_id SERIAL PRIMARY KEY,
    student_id VARCHAR(20) REFERENCES users(student_id) ON DELETE CASCADE,
    booth_id INTEGER REFERENCES booths(booth_id) ON DELETE CASCADE,
    acquired_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    method VARCHAR(10) DEFAULT 'QR',
    CONSTRAINT unique_stamp_per_user_booth UNIQUE (student_id, booth_id)
);

-- ========================================
-- 5. 인덱스 생성
-- ========================================
CREATE INDEX idx_stamps_student_id ON stamps(student_id);
CREATE INDEX idx_stamps_booth_id ON stamps(booth_id);
CREATE INDEX idx_stamps_acquired_at ON stamps(acquired_at);
CREATE INDEX idx_booths_event_id ON booths(event_id);
CREATE INDEX idx_users_phone_number ON users(phone_number);
CREATE INDEX idx_users_role ON users(role);

FROM stamps s
JOIN users u ON s.student_id = u.student_id
JOIN booths b ON s.booth_id = b.booth_id
ORDER BY s.acquired_at;
